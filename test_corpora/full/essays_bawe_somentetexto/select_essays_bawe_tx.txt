#!/usr/bin/env python3 """Select and copy files containing 'Essay' in their filename (case-insensitive) from the current `bawe_work` tree into a new folder `bawe_essays`. Usage: python3 select_essays_bawe.py [--root /abs/path/to/bawe_work] [--dry-run] [--ext .txt .docx] By default the script assumes it's located in the `bawe_work` folder and will create `bawe_essays` inside the same folder. """ from __future__ import annotations import argparse import shutil from pathlib import Path from typing import List def find_essay_files(root: Path, exts: List[str] | None = None) -> List[Path]: """Recursively find files whose name contains 'essay' (case-insensitive). If exts is provided, only files with those extensions (case-insensitive) will be considered. """ matches: List[Path] = [] for p in root.rglob("*"): if not p.is_file(): continue name_lower = p.name.lower() if "essay" not in name_lower: continue if exts: if p.suffix.lower() not in [e.lower() for e in exts]: continue matches.append(p) return matches def unique_dest(dest_dir: Path, filename: str) -> Path: """Return a non-conflicting path under dest_dir for filename. If dest_dir/filename exists, append a numeric suffix before the extension. """ candidate = dest_dir / filename if not candidate.exists(): return candidate stem = Path(filename).stem suffix = Path(filename).suffix i = 1 while True: new_name = f"{stem}_{i}{suffix}" candidate = dest_dir / new_name if not candidate.exists(): return candidate i += 1 def copy_files(files: List[Path], dest_dir: Path, dry_run: bool = False) -> List[Path]: """Copy files to dest_dir. Return list of final destination paths (would-be when dry-run).""" dest_dir.mkdir(parents=True, exist_ok=True) copied: List[Path] = [] for src in files: dest = unique_dest(dest_dir, src.name) if dry_run: copied.append(dest) continue shutil.copy2(src, dest) copied.append(dest) return copied def main() -> int: parser = argparse.ArgumentParser(description="Collect BAWE essays into one folder.") parser.add_argument("--root", type=Path, default=Path(__file__).resolve().parent, help="Path to the bawe_work folder (defaults to script's folder)") parser.add_argument("--dry-run", action="store_true", help="Show what would be copied") parser.add_argument("--ext", nargs="*", help="Optional list of extensions to include (e.g. .txt .docx)") args = parser.parse_args() root: Path = args.root if not root.exists() or not root.is_dir(): print(f"ERROR: root path does not exist or is not a directory: {root}") return 2 dest = root / "bawe_essays" exts = args.ext if args.ext else None print(f"Searching under: {root}") files = find_essay_files(root, exts) print(f"Found {len(files)} file(s) with 'Essay' in the name.") if len(files) == 0: print("Nothing to copy.") return 0 print(f"Destination folder: {dest}") copied = copy_files(files, dest, dry_run=args.dry_run) if args.dry_run: print("Dry-run mode: no files were copied. Example destinations:") else: print("Files copied:") for src, dst in zip(files, copied): print(f"- {src} -> {dst}") print(f"Total: {len(copied)} files {'would be copied' if args.dry_run else 'copied'}.") return 0 if __name__ == "__main__": raise SystemExit(main())
